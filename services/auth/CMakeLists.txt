solo_api(auth)
find_program(ODB_COMPILER odb REQUIRED)
find_library(ODB_LIBRARY odb REQUIRED)
find_library(ODB_PGSQL_LIBRARY odb-pgsql REQUIRED)
find_library(ODB_BOOST_LIBRARY odb-boost REQUIRED)
set(ODB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src/generated/odb)
set(ODB_INPUT ${CMAKE_CURRENT_SOURCE_DIR}/src/models/entities.hpp)
add_custom_command(OUTPUT ${ODB_DIR}/entities-odb.cxx ${ODB_DIR}/entities-odb.hxx ${ODB_DIR}/entities-odb.ixx
  COMMAND ${CMAKE_COMMAND} -E make_directory ${ODB_DIR}
  COMMAND ${ODB_COMPILER} --std c++11 --database pgsql --profile boost/date-time --generate-query
    --output-dir ${ODB_DIR} -I ${CMAKE_CURRENT_SOURCE_DIR}/src/models ${ODB_INPUT}
  DEPENDS ${ODB_INPUT} VERBATIM)
add_library(auth-logic src/logic/auth.cpp)
add_dependencies(auth-logic auth-api)
target_include_directories(auth-logic PUBLIC src)
target_link_libraries(auth-logic PUBLIC solo-runtime)
add_library(auth-odb src/logic/odb_repository.cpp ${ODB_DIR}/entities-odb.cxx)
target_include_directories(auth-odb PUBLIC src ${PROJECT_SOURCE_DIR}/libs PRIVATE src/models /usr/include/postgresql)
target_link_libraries(auth-odb PUBLIC ${ODB_LIBRARY} ${ODB_PGSQL_LIBRARY} ${ODB_BOOST_LIBRARY})
add_executable(auth-service src/main.cpp src/views/health_ready.cpp src/views/healthz.cpp src/views/ready.cpp src/views/auth_register.cpp src/views/auth_login.cpp src/views/auth_me.cpp src/views/auth_logout.cpp src/views/internal_verify.cpp)
target_link_libraries(auth-service PRIVATE auth-logic auth-odb)
install(TARGETS auth-service RUNTIME DESTINATION bin)
if(SOLO_BUILD_TESTS)
  add_executable(auth-pool-probe tests/pool_probe.cpp)
  target_include_directories(auth-pool-probe PRIVATE src/models /usr/include/postgresql)
  target_link_libraries(auth-pool-probe PRIVATE auth-odb pq)
  add_executable(auth-unit-probe tests/unit_probe.cpp)
  target_link_libraries(auth-unit-probe PRIVATE auth-logic)
endif()
