cmake_minimum_required(VERSION 3.16)
project(soloapp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SOLO_SERVICE "all" CACHE STRING "auth, whoami or all")
option(SOLO_BUILD_TESTS "Build C++ unit-test probes for Python testsuite" ON)
find_package(Drogon CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
add_subdirectory(libs/runtime)
include(libs/api_codegen/Codegen.cmake)
if(SOLO_SERVICE STREQUAL "all" OR SOLO_SERVICE STREQUAL "auth")
  add_subdirectory(services/auth)
endif()
if(SOLO_SERVICE STREQUAL "all" OR SOLO_SERVICE STREQUAL "whoami")
  add_subdirectory(services/whoami)
endif()

if(SOLO_SERVICE STREQUAL "all" OR SOLO_SERVICE STREQUAL "client-notify")
  add_subdirectory(services/client-notify)
endif()
